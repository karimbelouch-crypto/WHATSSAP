// Vercel serverless function: sends one WhatsApp message (template, or free-form text/image within the 24h window) via Meta Cloud API. // Env vars (Vercel > Settings > Environment Variables): // WA_TOKEN Meta access token (permanent) // WA_PHONE_ID WhatsApp phone number ID // APP_PASSWORD password that protects this endpoint // GRAPH_VERSION optional, default v23.0 module.exports = async function handler(req, res) { if (req.method !== "POST") return res.status(405).json({ error: "POST only" }); if (!process.env.APP_PASSWORD || req.headers["x-password"] !== process.env.APP_PASSWORD) { return res.status(401).json({ error: "mot de passe ghalat" }); } if (!process.env.WA_TOKEN || !process.env.WA_PHONE_ID) { return res.status(500).json({ error: "WA_TOKEN wla WA_PHONE_ID naqsin f Vercel" }); } const { to, template, lang, name, mediaType, mediaUrl, message } = req.body || {}; const phone = String(to || "").replace(/\D/g, ""); if (phone.length < 9 || (!template && !message)) return res.status(400).json({ error: "numero wla message naqes" }); let payload; if (template) { const components = []; if (mediaType && mediaUrl) { const key = mediaType === "video" ? "video" : "image"; components.push({ type: "header", parameters: [{ type: key, [key]: { link: mediaUrl } }] }); } if (name) components.push({ type: "body", parameters: [{ type: "text", text: String(name) }] }); payload = { messaging_product: "whatsapp", to: phone, type: "template", template: { name: template, language: { code: lang || "fr" }, components }, }; } else if (mediaUrl) { const key = mediaType === "video" ? "video" : "image"; payload = { messaging_product: "whatsapp", to: phone, type: key, [key]: { link: mediaUrl, caption: String(message || "") } }; } else { payload = { messaging_product: "whatsapp", to: phone, type: "text", text: { body: String(message) } }; } try { const v = process.env.GRAPH_VERSION || "v23.0"; const r = await fetch(`https://graph.facebook.com/${v}/${process.env.WA_PHONE_ID}/messages`, { method: "POST", headers: { Authorization: `Bearer ${process.env.WA_TOKEN}`, "Content-Type": "application/json" }, body: JSON.stringify(payload), }); const j = await r.json(); if (!r.ok) return res.status(r.status).json({ error: (j.error && j.error.message) || "Meta error" }); return res.status(200).json({ ok: true, id: j.messages && j.messages[0] && j.messages[0].id }); } catch (e) { return res.status(502).json({ error: "ma-9drnach nwslo l Meta" }); } };
